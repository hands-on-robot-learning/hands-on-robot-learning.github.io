import { parseYaml } from '../../lib/yaml';
import WeekCard from '../../components/WeekCard.astro';
import scheduleYaml from '../../data/schedule.yaml?raw';

export const weeks = (parseYaml(scheduleYaml).weeks ?? []);

export const tagRules = [
	['hardware', ['hardware', 'robotics essentials', 'safety', 'assembly', 'calibration', 'so-101', 'wake the robot', 'robot kit']],
	['teleop', ['teleoperation', 'teleop', 'leader-follower', 'leader arm', 'keyboard', 'gamepad']],
	['data', ['data', 'dataset', 'demonstration', 'episode', 'metadata', 'camera']],
	['baselines', ['baseline', 'handcrafted', 'segmentation', 'homography', 'scripted', 'code-as-policies']],
	['imitation learning', ['imitation', 'behavior cloning', 'action chunking', 'act', 'diffusion', 'policy training']],
	['evaluation', ['evaluation', 'leaderboard', 'trial', 'failure', 'ablation', 'robustness']],
	['VLA', ['vla', 'vision-language-action', 'smolvla', 'language', 'generalist']],
	['showcase', ['showcase', 'demo', 'final demos']],
];

export const inferTags = (week) => {
	const text = [
		week.theme,
		week.lecture_title,
		week.lab_focus,
		week.milestone,
		...(week.core_topics ?? []),
	]
		.join(' ')
		.toLowerCase();

	const tags = tagRules
		.filter(([, terms]) => terms.some((term) => text.includes(term)))
		.map(([tag]) => tag);

	return tags.length ? tags.slice(0, 4) : ['systems'];
};
